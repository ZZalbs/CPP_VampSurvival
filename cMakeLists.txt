cmake_minimum_required(VERSION 3.20)
project(ShapeSurvivor LANGUAGES CXX)
find_package(raylib CONFIG REQUIRED)
add_executable(ShapeSurvivor src/main.cpp)
target_compile_features(ShapeSurvivor PRIVATE cxx_std_17)
target_link_libraries(ShapeSurvivor PRIVATE raylib)
if(MSVC)
    target_compile_options(ShapeSurvivor PRIVATE /W4 /utf-8)
else()
    target_compile_options(ShapeSurvivor PRIVATE -Wall -Wextra -Wpedantic)
endif()
